# ⭐ 1. SQL-READY DATABASE SCHEMA

# ================================================

Below is a relational schema suitable for PostgreSQL.

```sql
-- ===============================
-- Standards
-- ===============================
CREATE TABLE standards (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    version VARCHAR(50),
    category VARCHAR(100),
    description TEXT
);

-- ===============================
-- Standard Controls
-- ===============================
CREATE TABLE standard_controls (
    id SERIAL PRIMARY KEY,
    standard_id INT REFERENCES standards(id),
    code VARCHAR(50),
    title VARCHAR(255),
    description TEXT
);

-- ===============================
-- Master Controls
-- ===============================
CREATE TABLE master_controls (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    category VARCHAR(100),
    subcategory VARCHAR(100),
    description TEXT,
    priority INT
);

-- ===============================
-- Control Mapping
-- ===============================
CREATE TABLE control_mappings (
    id SERIAL PRIMARY KEY,
    master_control_id INT REFERENCES master_controls(id),
    standard_control_id INT REFERENCES standard_controls(id),
    equivalence_type VARCHAR(50) -- full, partial, related
);

-- ===============================
-- Organizations (Tenants)
-- ===============================
CREATE TABLE organizations (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    industry VARCHAR(100),
    size INT,
    domain VARCHAR(255),
    created_at TIMESTAMP DEFAULT NOW()
);

-- ===============================
-- Users
-- ===============================
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    org_id INT REFERENCES organizations(id),
    name VARCHAR(255),
    email VARCHAR(255) UNIQUE,
    role VARCHAR(50), -- admin, auditor, user, consultant
    password_hash TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);

-- ===============================
-- Evidence
-- ===============================
CREATE TABLE evidence (
    id SERIAL PRIMARY KEY,
    org_id INT REFERENCES organizations(id),
    master_control_id INT REFERENCES master_controls(id),
    file_url TEXT,
    source VARCHAR(100),
    status VARCHAR(50), -- valid, pending, failed
    uploaded_at TIMESTAMP DEFAULT NOW()
);

-- ===============================
-- Integrations
-- ===============================
CREATE TABLE integrations (
    id SERIAL PRIMARY KEY,
    org_id INT REFERENCES organizations(id),
    provider VARCHAR(100),
    status VARCHAR(50),
    config JSONB,
    last_sync TIMESTAMP
);

-- ===============================
-- Automated Checks
-- ===============================
CREATE TABLE automated_checks (
    id SERIAL PRIMARY KEY,
    master_control_id INT REFERENCES master_controls(id),
    provider VARCHAR(100),
    endpoint VARCHAR(255),
    pass_criteria JSONB,
    fail_criteria JSONB
);

-- ===============================
-- Risk Register
-- ===============================
CREATE TABLE risks (
    id SERIAL PRIMARY KEY,
    org_id INT REFERENCES organizations(id),
    asset VARCHAR(255),
    threat TEXT,
    vulnerability TEXT,
    impact INT,
    likelihood INT,
    score INT,
    treatment TEXT,
    status VARCHAR(50),
    created_at TIMESTAMP DEFAULT NOW()
);

-- ===============================
-- Internal Audit Findings
-- ===============================
CREATE TABLE audit_findings (
    id SERIAL PRIMARY KEY,
    org_id INT REFERENCES organizations(id),
    master_control_id INT REFERENCES master_controls(id),
    severity VARCHAR(50),
    finding TEXT,
    corrective_action TEXT,
    status VARCHAR(50),
    created_at TIMESTAMP DEFAULT NOW()
);
```

```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: backend

  replicas: 2

      app: backend
  template:
    metadata:
      labels:
        app: backend
    spec:
      containers:
        - name: backend
          image: compliance_backend:latest
          ports:
            - containerPort: 5000
          envFrom:
            - secretRef:
                name: backend-secrets
```

**service.yaml**

```yaml
apiVersion: v1
kind: Service
metadata:
  name: backend-service
spec:
  selector:
    app: backend
  ports:
    - protocol: TCP
      port: 5000
      targetPort: 5000
  type: ClusterIP
```
