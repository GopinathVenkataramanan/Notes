### Here is the **updated code snippet** (only changed or new parts):

```jsx
// Add imports
import axios from "axios";  // npm install axios

// Add state to hold backend response
const [generationResponse, setGenerationResponse] = useState(null);
const [isGenerating, setIsGenerating] = useState(false);
const [generationError, setGenerationError] = useState(null);

const onGenerateCode = async () => {
  setIsGenerating(true);
  setGenerationError(null);
  setGenerationResponse(null);

  // Prepare payload: nodes + edges + configs
  const nodes = elements.filter((el) => !el.source).map(({ id, data, position }) => ({
    id,
    type: data.label,
    position,
    config: data.config,
  }));

  const edges = elements.filter((el) => el.source).map(({ id, source, target }) => ({
    id,
    source,
    target,
  }));

  const payload = { nodes, edges };

  try {
    // Replace with your backend API endpoint
    const response = await axios.post("http://localhost:4000/api/generate", payload);
    setGenerationResponse(response.data);
  } catch (err) {
    setGenerationError(err.message || "Unknown error");
  } finally {
    setIsGenerating(false);
  }
};

return (
  <>
    {/* ... your existing JSX ... */}

    {/* Add generate button below sidebar or anywhere you prefer */}
    <aside style={{ padding: 10, borderTop: "1px solid #ddd" }}>
      <button onClick={onGenerateCode} disabled={isGenerating}>
        {isGenerating ? "Generating..." : "Generate Code"}
      </button>

      {generationResponse && (
        <pre style={{ whiteSpace: "pre-wrap", marginTop: 10, maxHeight: 200, overflow: "auto" }}>
          {JSON.stringify(generationResponse, null, 2)}
        </pre>
      )}

      {generationError && (
        <div style={{ color: "red", marginTop: 10 }}>Error: {generationError}</div>
      )}
    </aside>

    {/* ... rest of your JSX ... */}
  </>
);
```
